generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")
}

enum TaskStatus {
  PENDING
  DONE
  DISMISSED
}

enum InterpretationStatus {
  PENDING
  CONFIRMED
  FAILED
}

enum NotificationStatus {
  SENT
  FAILED
}

enum NotificationActionType {
  DONE
  TOMORROW
  DISMISS
}

enum TaskPriority {
  LOW
  MEDIUM
  HIGH
}

enum Plan {
  FREE
  PAID
}

enum NotificationChannel {
  EMAIL
  TELEGRAM
  BOTH
}

model User {
  id                  String              @id @default(uuid())
  email               String              @unique
  passwordHash        String?
  googleId            String?             @unique
  name                String?
  telegramChatId      String?             @unique
  plan                Plan                @default(FREE)
  notificationChannel NotificationChannel @default(EMAIL)
  quietHoursStart     String?
  quietHoursEnd       String?
  timeZone            String              @default("America/Argentina/Buenos_Aires")
  dailySummaryTime    String              @default("08:00")
  lastSummaryOn       String?
  messagesUsed        Int                 @default(0)
  usagePeriod         String?
  googleCalendarRefreshToken String?
  googleCalendarConnectedAt  DateTime?
  createdAt           DateTime            @default(now())
  updatedAt           DateTime            @updatedAt

  tasks Task[]
  interpretations Interpretation[]
  notifications NotificationLog[]

  @@map("users")
}

model Category {
  id        String   @id @default(uuid())
  name      String   @unique
  color     String?
  createdAt DateTime @default(now())

  tasks Task[]

  @@map("categories")
}

model Task {
  id           String       @id @default(uuid())
  userId       String
  categoryId   String?
  interpretationId String?

  rawText      String
  title        String
  dueDate      DateTime?
  reminderAt   DateTime?
  notifiedAt   DateTime?
  googleEventId String?
  status       TaskStatus   @default(PENDING)
  priority     TaskPriority @default(MEDIUM)
  aiConfidence Float?

  createdAt    DateTime     @default(now())
  updatedAt    DateTime     @updatedAt

  user         User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  category     Category?    @relation(fields: [categoryId], references: [id], onDelete: SetNull)
  interpretation Interpretation? @relation(fields: [interpretationId], references: [id], onDelete: SetNull)
  notifications NotificationLog[]
  actionTokens NotificationActionToken[]

  @@map("tasks")
}

model Interpretation {
  id           String               @id @default(uuid())
  userId       String
  originalText String
  parsedTasks  Json?
  status       InterpretationStatus @default(PENDING)
  error        String?
  createdAt    DateTime             @default(now())
  confirmedAt  DateTime?

  user  User   @relation(fields: [userId], references: [id], onDelete: Cascade)
  tasks Task[]

  @@map("interpretations")
}

model NotificationLog {
  id        String             @id @default(uuid())
  userId    String
  taskId    String?
  channel   NotificationChannel
  status    NotificationStatus
  subject   String
  body      String
  error     String?
  sentAt    DateTime           @default(now())

  user User  @relation(fields: [userId], references: [id], onDelete: Cascade)
  task Task? @relation(fields: [taskId], references: [id], onDelete: SetNull)

  @@map("notification_logs")
}

model NotificationActionToken {
  id        String                 @id @default(uuid())
  tokenHash String                 @unique
  taskId    String
  action    NotificationActionType
  expiresAt DateTime
  usedAt    DateTime?
  createdAt DateTime               @default(now())

  task Task @relation(fields: [taskId], references: [id], onDelete: Cascade)

  @@map("notification_action_tokens")
}
